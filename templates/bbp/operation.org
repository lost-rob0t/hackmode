Export in-scope targets to =.config/targets.txt=:
#+begin_src elisp :var tbl=scope :results silent
(let* ((dir (expand-file-name ".config"))
       (file (expand-file-name "targets.txt" dir))
       (in-scope
        (cl-loop for row in tbl
                 when (and (consp row)
                           (>= (length row) 2)
                           (member (downcase (format "%s" (nth 1 row)))
                                   '("true" "yes" "y")))
                 collect (format "%s" (nth 0 row)))))
  (make-directory dir t)
  (with-temp-file file
    (insert (mapconcat #'identity in-scope "\n"))))
#+end_src
